\section{Governing equations}

For a uniform transmission line with series impedance per unit length
$Z'=R+\jmath\omega L$ and shunt admittance per unit length
$Y'=G+\jmath\omega C$,
\begin{equation}
Z_0=\sqrt{\frac{Z'}{Y'}},
\qquad
\gamma=\sqrt{Z'Y'}=\alpha+\jmath\beta,
\qquad
\lambda=\frac{2\pi}{\beta}.
\end{equation}
For a lossless matched line, $\Gamma_L=0$, so
\begin{equation}
\phasor V(z)=V_0^+\ee^{-\jmath\beta z},
\qquad
\phasor I(z)=\frac{V_0^+}{Z_0}\ee^{-\jmath\beta z}.
\end{equation}
The time convention is $\ee^{\jmath\omega t}$, and all sinusoidal amplitudes
below are peak values.

\section{Problem 2: Coaxial cable}

\subsection{Exact evaluation}

At $500~\mathrm{MHz}$,
\[
\omega=3.141592654\times10^9~\mathrm{rad/s},
\quad
Z'=4+\jmath785.398163,
\quad
Y'=0.001+\jmath0.314159265.
\]
Therefore
\begin{align}
Z_0&=50.0001748-\jmath0.0477458~\Omega
=50.0001976\angle(-0.0547125\degree)~\Omega,\\
\gamma&=0.06499997+\jmath15.70797043~\mathrm{m}^{-1}.
\end{align}
The exact attenuation in decibels is
\[
\alpha_{\mathrm{dB}}=8.6858896(0.06499997)
=0.5645897~\mathrm{dB/m}.
\]

\subsection{Low-loss calculation}

The conditions are satisfied because
\[
R=4\ll\omega L=785.398,
\qquad
G=0.001\ll\omega C=0.314159.
\]
Consequently,
\begin{align*}
Z_0&\simeq\sqrt{L/C}
=\sqrt{250\times10^{-9}/(100\times10^{-12})}=50~\Omega,\\
\alpha&\simeq\frac{1}{2}\left(\frac{4}{50}
+(0.001)(50)\right)=0.065~\mathrm{Np/m},\\
\beta&\simeq(3.141592654\times10^9)
\sqrt{(250\times10^{-9})(100\times10^{-12})}\\
&=15.70796327~\mathrm{rad/m}.
\end{align*}

\section{Problem 3: Three-wavelength matched line}

\MatchedLineDiagram{v_s(t)=5\cos(2\pi\,2\!\times\!10^8t)}{75~\Omega}{50~\Omega}{3~\mathrm{m}}{R_L=50~\Omega}

\subsection{Wave amplitude at the line input}

Matching gives $Z_{\mathrm{in}}=Z_0=50~\Omega$.  The source divider is
\[
V_0^+=V_s\frac{Z_0}{R_s+Z_0}
=5\frac{50}{75+50}=2~\mathrm{V},
\qquad I_0^+=\frac{2}{50}=0.04~\mathrm{A}.
\]

\subsection{Space-time solution}

Since
\[
\lambda=\frac{2\times10^8}{2\times10^8}=1~\mathrm{m},
\qquad \beta=2\pi~\mathrm{rad/m},
\qquad \omega=4\pi\times10^8~\mathrm{rad/s},
\]
the required instantaneous quantities are
\begin{align}
v(z,t)&=2\cos(4\pi\times10^8t-2\pi z)~\mathrm{V},\\
i(z,t)&=0.04\cos(4\pi\times10^8t-2\pi z)~\mathrm{A}.
\end{align}
The voltage-current ratio is $2/0.04=50~\Omega$, as it must be for a pure
forward wave.

\subsection{Line length and phase at the load}

\[
N_\lambda=\frac{3}{1}=3,
\qquad
\beta\ell=(2\pi)(3)=6\pi~\mathrm{rad}.
\]
At $z=3~\mathrm{m}$,
\[
v_L(t)=2\cos(4\pi\times10^8t-6\pi)
=2\cos(4\pi\times10^8t)~\mathrm{V}.
\]
Thus the load and source sinusoids are in phase; the load peak amplitude is
$2/5=0.4$ times the ideal source amplitude because of the source resistance.

\answer{The line length is $3\lambda$ ($1080\degree$).  The load voltage is
$2~\mathrm{V}$ peak at the same phase as the source.}

\section{Problem 4: One-wavelength air line}

\MatchedLineDiagram{v_s(t)=10\cos(2\pi\,5\!\times\!10^7t)}{25~\Omega}{100~\Omega}{6~\mathrm{m}}{R_L=100~\Omega}

\subsection{Matched termination}

The frequency is $50~\mathrm{MHz}$.  In air,
\[
\lambda=\frac{3\times10^8}{5\times10^7}=6~\mathrm{m},
\quad
\beta=\frac{2\pi}{6}=\frac{\pi}{3}~\mathrm{rad/m},
\quad
\omega=\pi\times10^8~\mathrm{rad/s}.
\]
Since the load is matched,
\[
V_0^+=10\frac{100}{25+100}=8~\mathrm{V},
\qquad I_0^+=8/100=0.08~\mathrm{A}.
\]
Hence, for $0\le z\le6~\mathrm{m}$,
\begin{align}
v(z,t)&=8\cos\left(\pi\times10^8t-\frac{\pi}{3}z\right)~\mathrm{V},\\
i(z,t)&=0.08\cos\left(\pi\times10^8t-\frac{\pi}{3}z\right)~\mathrm{A}.
\end{align}
The average power is constant everywhere on a lossless matched line:
\begin{equation}
P_{\mathrm{av}}=\frac{1}{2}V_0^+I_0^+
=\frac{1}{2}(8)(0.08)=0.320~\mathrm{W}.
\end{equation}

\subsection{Open-circuit termination}

The matched expression cannot be retained.  For an open circuit,
$\Gamma_L=+1$, so $|V^-|=|V^+|$.  Since the line is exactly one wavelength,
the open circuit is repeated at the input: $Z_{\mathrm{in}}\to\infty$.
No current flows through $R_s$, and the input voltage has the full
$10~\mathrm{V}$ peak value.  Therefore $V_0^+=V_0^-=5~\mathrm{V}$ at $z=0$,
and
\begin{align}
v(z,t)&=\Re\left\{5\left(\ee^{-\jmath\beta z}
+\ee^{\jmath\beta z}\right)\ee^{\jmath\omega t}\right\}\\
&=10\cos\left(\frac{\pi z}{3}\right)
\cos(\pi\times10^8t)~\mathrm{V}.
\end{align}
This result has fixed voltage nodes and antinodes, which identifies it as a
standing wave.  At the load $z=6~\mathrm{m}$, the current is zero and
$v_L(t)=10\cos(\pi\times10^8t)~\mathrm{V}$.

\answer{Matched case: $P_L=0.320~\mathrm{W}$.  Open-circuit case: the
single-wave equation is invalid and must be replaced by the superposition of
incident and reflected waves.}
